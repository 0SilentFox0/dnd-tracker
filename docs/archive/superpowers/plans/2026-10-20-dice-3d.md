```ts
import { diceSlots } from "@/lib/utils/common/dice";
import type { DiceRollGroup, DieSides, RolledDie } from "@/types/battle";

const SIDES = new Set<number>([4, 6, 8, 10, 12, 20]);

const isSides = (n: number): n is DieSides => SIDES.has(n);

export function attackDiceRolls(input: { attackerId: string; targetId: string; d20: number; second?: { value: number; serverRolled: boolean }; chosenD20?: number; damage: number[]; damageFormula: string }): DiceRollGroup[] {
  const { attackerId: rollerId, targetId, second } = input;

  const firstChosen = !!second && input.chosenD20 === input.d20;

  const d20s: RolledDie[] = [{ sides: 20, value: input.d20, ...(firstChosen && { chosen: true }) }];

  if (second) d20s.push({ sides: 20, value: second.value, ...(!firstChosen && { chosen: true }), ...(second.serverRolled && { server: true }) });

  const slots = diceSlots(input.damageFormula);

  const damage = input.damage.flatMap((value, i): RolledDie[] => {
    const sides = slots[i];

    return sides !== undefined && isSides(sides) ? [{ sides, value }] : [];
  });

  return [{ kind: "attack", rollerId, targetId, dice: d20s }, ...(damage.length > 0 ? [{ kind: "damage" as const, rollerId, targetId, dice: damage }] : [])];
}

export function spellDiceRolls(input: { casterId: string; sides: number; rolls: number[]; heal: boolean; saves: Array<{ participantId: string; roll: number }>; presetSaveIds: ReadonlySet<string> }): DiceRollGroup[] {
  const { sides } = input;

  const power: DiceRollGroup[] = isSides(sides) && input.rolls.length > 0 ? [{ kind: input.heal ? "heal" : "damage", rollerId: input.casterId, dice: input.rolls.map((value) => ({ sides, value })) }] : [];

  const saves = input.saves.map((s): DiceRollGroup => ({ kind: "save", rollerId: s.participantId, dice: [{ sides: 20, value: s.roll, ...(!input.presetSaveIds.has(s.participantId) && { server: true }) }] }));

  return [...power, ...saves];
}

export function moraleDiceRolls(participantId: string, d10: number): DiceRollGroup[] {
  return [{ kind: "morale", rollerId: participantId, dice: [{ sides: 10, value: d10 }] }];
}
```

Note the second-d20 `chosen` rule: when `chosenD20` equals the first die (including ties) only the first is chosen; otherwise the second.

- [ ] **Step 5: Implement `layout.ts`**

```ts
import type { DieSides } from "@/types/battle";

export const MAX_DICE = 12;

export const MAX_SAVES_SHOWN = 6;

const ROW_MAX = 6;

export interface DieSlot { sides: DieSides; value: number; chosen?: boolean; dropped?: boolean; row: number; col: number; rowLength: number }

type Die = Omit<DieSlot, "row" | "col" | "rowLength">;

export function layoutDice(groups: Die[][]): { slots: DieSlot[]; overflow: number } {
  const total = groups.reduce((a, g) => a + g.length, 0);

  let left = MAX_DICE;

  const rows: Die[][] = [];

  for (const group of groups) {
    const kept = group.slice(0, Math.max(0, left));

    left -= kept.length;

    for (let i = 0; i < kept.length; i += ROW_MAX) rows.push(kept.slice(i, i + ROW_MAX));
  }

  const slots = rows.flatMap((row, r) => row.map((die, c) => ({ ...die, row: r, col: c, rowLength: row.length })));

  return { slots, overflow: total - slots.length };
}
```

```ts
import { MAX_SAVES_SHOWN } from "./layout";

import type { BattleAction, DiceRollGroup, DieSides } from "@/types/battle";

export interface DiceThrowSpec {
  seed: number;
  groups: Array<Array<{ sides: DieSides; value: number; chosen?: boolean; dropped?: boolean }>>;
  caption: string;
  extraSaves: number;
}

const toDie = (group: DiceRollGroup) => {
  const pick = group.dice.some((d) => d.chosen);

  return group.dice.map((d) => ({ sides: d.sides, value: d.value, ...(d.chosen && { chosen: true }), ...(pick && !d.chosen && { dropped: true }) }));
};

function captionOf(action: BattleAction): string {
  const target = action.targets[0]?.participantName;

  if (action.actionType === "spell") return `${action.actorName} · ${action.actionDetails.spellName ?? "заклинання"}`;

  if (action.actionType === "attack") return `${action.actorName} · атака${target ? ` → ${target}` : ""}`;

  return `${action.actorName} · мораль`;
}

export function diceThrowOf(action: BattleAction): DiceThrowSpec | null {
  const rolls = action.actionDetails.diceRolls;

  if (action.isCancelled || !rolls?.length) return null;

  const saves = rolls.filter((g) => g.kind === "save");

  const shownSaves = saves.slice(0, MAX_SAVES_SHOWN).flatMap(toDie);

  const groups = [...rolls.filter((g) => g.kind !== "save").map(toDie), ...(shownSaves.length ? [shownSaves] : [])];

  return { seed: action.actionIndex, groups, caption: captionOf(action), extraSaves: Math.max(0, saves.length - MAX_SAVES_SHOWN) };
}
```

```tsx
import { forwardRef, useEffect, useImperativeHandle, useRef } from "react";

import type { DiceEngine } from "./engine";
import { DICE_TOTAL_MS } from "./engine";

import type { DiceThrowSpec } from "@/lib/utils/battle/fx/dice";

export interface DiceStageHandle {
  play(spec: DiceThrowSpec): Promise<void>;
  cancel(): void;
}

export { DICE_TOTAL_MS };

export const loadDiceScene = () => import("./scene");

// WebGL-шар монтується при першому кидку й живе до виходу з бою: один рендерер на бій
export const DiceStage = forwardRef<DiceStageHandle>(function DiceStage(_, ref) {
  const host = useRef<HTMLDivElement>(null);

  const engine = useRef<Promise<DiceEngine | null> | null>(null);

  const ensure = () => {
    engine.current ??= loadDiceScene()
      .then((m) => (host.current ? m.createThreeDice(host.current) : null))
      .catch((e: unknown) => {
        console.warn("dice stage", e);

        return null;
      });

    return engine.current;

    async play(spec) {
      const e = await ensure();

      if (e) await e.roll(spec);
    },
    cancel() {
      void engine.current?.then((e) => e?.cancel());
    },
  }));

  useEffect(() => () => void engine.current?.then((e) => e?.dispose()), []);

  return <div ref={host} className="fx-dice-layer" aria-hidden />;
});
```
